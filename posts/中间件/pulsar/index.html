mac 安装Pulsar#

参考官网：
https://pulsar.apache.org/docs/4.1.x/getting-started-standalone/

第一步，下载#

1

curl -LO "https://www.apache.org/dyn/closer.lua/pulsar/pulsar-4.1.2/apache-pulsar-4.1.2-bin.tar.gz?action=download"

解压：

1

tar xvfz apache-pulsar-4.1.2-bin.tar.gz

切换到根目录：

1

cd apache-pulsar-4.1.2

第二步，启动#

1

bin/pulsar standalone

第三步，创建topic#

再开一个终端窗口，运行如下命令：

1

bin/pulsar-admin topics create persistent://public/default/my-topic

第四步，生产消息（生产者）#

1

bin/pulsar-client produce my-topic --messages 'Hello Pulsar!'

第五步：消费消息（消费者）#

再开一个终端窗口，运行如下命令：

1

bin/pulsar-client consume my-topic -s 'my-subscription' -p Earliest -n 0

你会看到上一步中产生的消息：

1
2

----- got message -----
key:[null], properties:[], content:Hello Pulsar!

写更多消息：

1

bin/pulsar-client produce my-topic --messages "$(seq -s, -f 'Message NO.%g' 1 10)"

@Component
@Slf4j
public class PulsarProducer {

    private final PulsarTemplate<MessageDto> pulsarTemplate;

    // 注入 PulsarTemplate，并指定消息类型
    public PulsarProducer(PulsarTemplate<MessageDto> pulsarTemplate) {
        this.pulsarTemplate = pulsarTemplate;
    }

    /**
     * 发送消息到指定的 Topic
     * @param message   消息内容
     */
    public void sendMessage(MessageDto message) {
        try {
            // 使用 send 方法发送消息
            pulsarTemplate.send(TopicConstant.MY_TOPIC, message);
            log.info("消息发送成功到 Topic: {}, Message: {}", TopicConstant.MY_TOPIC, message);
        } catch (Exception e) {
            log.error("消息发送失败: {}", e.getMessage());
        }
    }
}

@Component
@Slf4j
public class PulsarConsumer {

    /**
     * 使用 @PulsarListener 监听指定的 Topic
     * @param message 接收到的消息，Spring 会自动反序列化为 MessageDto 对象
     */
    @PulsarListener(topics = TopicConstant.MY_TOPIC, subscriptionName = "my-subscription")
    public void listen(MessageDto message) {
        log.info("接收到消息: {}", message);
        // 在这里添加你的业务逻辑，例如处理数据、保存到数据库等
        System.out.println("成功处理消息: " + message.getContent());
    }
}

@RestController
@RequestMapping("/api/pulsar")
public class PulsarController {

    private final PulsarProducer pulsarProducer;

    public PulsarController(PulsarProducer pulsarProducer) {
        this.pulsarProducer = pulsarProducer;
    }

    @GetMapping("/send")
    public ResponseEntity<String> sendMessage(@RequestParam String message) {
        pulsarProducer.sendMessage(new MessageDto("1",message, LocalDateTime.now().toString()));
        return ResponseEntity.ok("消息已发送到 " + TopicConstant.MY_TOPIC);
    }
}

@Configuration
public class PulsarConfig {

    @Value("${spring.pulsar.client.service-url}")
    private String serviceUrl;

    @Bean
    public PulsarClient pulsarClient() throws PulsarClientException {
        ClientBuilder clientBuilder = PulsarClient.builder()
                .serviceUrl(serviceUrl)
                .operationTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
                .connectionTimeout(10, java.util.concurrent.TimeUnit.SECONDS);

        // 可以添加认证等其他配置
        // clientBuilder.authentication(AuthenticationFactory.token("your-token"));

        return clientBuilder.build();
    }
}
